ingredients:
  - item: "Egg Whites (6 large eggs)"
    amount: "250g"
  - item: "Salt"
    amount: "pinch"
  - item: "Caster Sugar"
    amount: "300g"
  - item: "Potato Starch"
    amount: "1 tsp"
  - item: "Vinegar or Lemon Juice"
    amount: "1 tsp"
  - item: "Double Cream"
    amount: "450ml"
  - item: "Mascarpone"
    amount: "250g"
  - item: "Berry Smoothie Mixture"
    amount: "100ml"
  - item: "Fruit (for filling and topping)"
    amount: "100g"
instructions:
  - step: "Whip Egg Whites, Salt, gradually add Sugar to stiff peaks. Fold in Starch, Vinegar."
  - step: "Pipe onto lined tray. Bake at 110°C for 90 min."
  - step: "Whip Cream, fold in Mascarpone, Smoothie."
  - step: "Layer with Fruit, chill."
