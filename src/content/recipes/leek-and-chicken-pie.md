ingredients:
  - item: "Leeks"
    amount: "2"
  - item: "Sweet Potatoes"
    amount: "3"
  - item: "Chicken Breast"
    amount: "800g"
  - item: "Unsalted Butter"
    amount: "100g"
  - item: "Double Cream or Milk"
    amount: "200ml"
  - item: "Puff Pastry (or Pasta)"
    amount: "1 large pack (or 500g pasta)"
  - item: "Chicken Stock"
    amount: "15g (2-3 cubes)"
  - item: "Smoked Paprika"
    amount: "to taste"
  - item: "Rice"
    amount: "200g"
instructions:
  - step: "Prepare a wok and saucepan."
  - step: "Preheat oven to 200°C (180°C fan)."
  - step: "Dice Leeks, Sweet Potato and Chicken."
  - step: "Fry Leeks and Chicken in wok with Butter until browned."
  - step: "Fry Sweet Potato in saucepan with Oil for 5 minutes, then coat in Smoked Paprika and fry for 5 more minutes. Add to wok."
  - step: "Add Rice to saucepan, mix briefly, soften in a little boiling water for 5 minutes. Add to wok."
  - step: "Mix Chicken Stock with boiling water, pour into wok, simmer for 5 minutes."
  - step: "Stir in Double Cream, simmer until thickened."
  - step: "Pour mixture into baking dish. Cover with Puff Pastry, decorate."
  - step: "Bake for 10 minutes or until golden and crisp."
  - step: "Serve hot, optionally with Broccoli."
